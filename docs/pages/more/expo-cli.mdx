import React, { useEffect, useRef, useState } from "react";
import { View, Text, TouchableOpacity, StyleSheet, ActivityIndicator, Alert } from "react-native";
import { Camera } from "expo-camera";
import * as MediaLibrary from "expo-media-library";
import { detectIngredientFromImage } from "../services/visionApi";

export default function ScanScreen({ navigation }) {
  const [hasPermission, setHasPermission] = useState(null);
  const [isProcessing, setIsProcessing] = useState(false);
  const cameraRef = useRef(null);

  useEffect(() => {
    (async () => {
      const cameraStatus = await Camera.requestCameraPermissionsAsync();
      const mediaStatus = await MediaLibrary.requestPermissionsAsync();
      setHasPermission(cameraStatus.status === "granted");
    })();
  }, []);

  const takePicture = async () => {
    if (!cameraRef.current) return;
    try {
      setIsProcessing(true);
      const photo = await cameraRef.current.takePictureAsync({ base64: true, quality: 0.6 });
      // Send base64 to vision API
      const detected = await detectIngredientFromImage(photo.base64);
      setIsProcessing(false);
      if (!detected || detected.length === 0) {
        Alert.alert("No ingredient found", "Try again with a clearer photo or another ingredient.");
        return;
      }
      // navigation to Search/Results with detected ingredients
      navigation.navigate("Search", { scannedIngredients: detected });
    } catch (err) {
      setIsProcessing(false);
      Alert.alert("Error", err.message || "Failed to take picture.");
    }
  };

  if (hasPermission === null) return <View style={styles.center}><Text>Requesting camera permission...</Text></View>;
  if (hasPermission === false) return <View style={styles.center}><Text>No camera access</Text></View>;

  return (
    <View style={{ flex: 1 }}>
      <Camera style={{ flex: 1 }} ref={cameraRef} ratio="16:9"/>
      <View style={styles.footer}>
        <TouchableOpacity style={styles.captureBtn} onPress={takePicture} disabled={isProcessing}>
          {isProcessing ? <ActivityIndicator color="#fff" /> : <Text style={styles.btnText}>Capture</Text>}
        </TouchableOpacity>

        <TouchableOpacity style={styles.cancelBtn} onPress={() => navigation.goBack()}>
          <Text style={styles.cancelText}>Cancel</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}

export default function RecipeScreen({ route, navigation }) {
  const recipe = route.params?.recipe;
  const [saved, setSaved] = useState(false);

  useEffect(() => {
    checkSaved();
  }, []);

  const checkSaved = async () => {
    const raw = await AsyncStorage.getItem("saved_recipes");
    const list = raw ? JSON.parse(raw) : [];
    setSaved(list.some((r) => r.id === recipe.id));
  };

  const toggleSave = async () => {
    const raw = await AsyncStorage.getItem("saved_recipes");
    let list = raw ? JSON.parse(raw) : [];
    if (saved) {
      list = list.filter((r) => r.id !== recipe.id);
    } else {
      list.unshift(recipe);
    }
    await AsyncStorage.setItem("saved_recipes", JSON.stringify(list));
    setSaved(!saved);
  };

  const handleShare = async () => {
    try {
      await Share.share({ message: `${recipe.name}\n\nIngredients: ${recipe.ingredients.join(", ")}\n\nInstructions: ${recipe.instructions.slice(0,120)}...` });
    } catch (err) {
      console.warn(err);
    }
  };

  return (
    <SafeAreaView style={{ flex: 1 }}>
      <ScrollView style={{ padding: 16 }}>
        <Text style={{ fontSize: 24, fontWeight: "800" }}>{recipe.name}</Text>
        <View style={{ flexDirection: "row", marginTop: 8, justifyContent: "space-between" }}>
          <Text style={{ color: "#333" }}>⏱ {recipe.cook_time} min</Text>
          <Text style={{ color: "#333" }}>💪 {recipe.protein}g protein</Text>
          <Text style={{ color: "#333" }}>{recipe.calories} kcal</Text>
        </View>

        <View style={{ flexDirection: "row", marginTop: 12, gap: 8 }}>
          <TouchableOpacity style={styles.btn} onPress={toggleSave}><Text style={{ color: "#fff" }}>{saved ? "Saved" : "Save"}</Text></TouchableOpacity>
          <TouchableOpacity style={[styles.btn, { backgroundColor: "#6c757d" }]} onPress={handleShare}><Text style={{ color: "#fff" }}>Share</Text></TouchableOpacity>
          <TouchableOpacity style={[styles.btn, { backgroundColor: "#f0ad4e" }]} onPress={() => alert("Shopping list added (stub)")}>
            <Text style={{ color: "#fff" }}>Add to Shopping List</Text>
          </TouchableOpacity>
        </View>

        <Text style={{ marginTop: 18, fontWeight: "700" }}>Ingredients</Text>
        {recipe.ingredients.map((ing, i) => <Text key={i}>• {ing}</Text>)}

        <Text style={{ marginTop: 12, fontWeight: "700" }}>Instructions</Text>
        <Text style={{ marginTop: 4 }}>{recipe.instructions}</Text>
      </ScrollView>
    </SafeAreaView>
  );
}

export async function searchRecipes(query, filters) {
  // Simulate network delay
  await new Promise((res) => setTimeout(res, 500));

  // Very simple filter logic for demo
  const q = (query || "").toLowerCase();
  let results = fakeAllRecipes.filter(r =>
    q === "" || r.name.toLowerCase().includes(q) || r.ingredients.join(" ").toLowerCase().includes(q)
  );

  if (filters?.highProtein) {
    results = results.filter(r => r.protein >= 25);
  }
  if (filters?.lowCalorie) {
    results = results.filter(r => r.calories <= 400);
  }
  if (filters?.highCalorie) {
    results = results.filter(r => r.calories >= 600);
  }
  return results;
}
